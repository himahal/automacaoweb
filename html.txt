                  function ValidaParRel(){
                     FormataCampo( document.all.dataInicial, '99/99/9999' );
                     FormataCampo( document.all.dataFinal, '99/99/9999' );
                     if( document.all.campoInicial.disabled == false && document.all.campoInicial.value == '' ){
                         document.all.campoInicial.value = '0';
                     }
                     if( document.all.nrRoadInicial.value == '' ){
                         document.all.nrRoadInicial.value = '0';
                     }
                     if( !ValidaCampos() ){
                         return false;
                     }
                     if( DataMaior( document.all.dataInicial.value, document.all.dataFinal.value ) ){
                         alert( 'Data inicial maior que data final' );
                         document.all.dataInicial.select();
                         return false;
                     }
                     if( document.all.dataInicial.value.substring(3) != document.all.dataFinal.value.substring(3)
                       && '0302240000' == '0914000000' ) {
                         alert( 'O período escolhido deve estar dentro de apenas um mês' );
                         document.all.dataInicial.select();
                         return false;
                     }
                     if( !ComparaValor( 'campoInicial', 'campoFinal', document.all.opcaoRel[document.all.opcaoRel.selectedIndex].text ) ){
                         return false;
                     }
                     if( !ComparaValor( 'nrRoadInicial', 'nrRoadFinal', 'Nr.Road' ) ){
                         return false;
                     }
                     if( document.all.selecionouAS.checked == false && document.all.selecionouROTA.checked == false ){
                         alert( 'É obrigatorio informar uma Área de Roterização' );
                         document.all.selecionouROTA.focus();
                         return false;
                     }
                     var classeIni = document.all.classeInicial.value.charCodeAt( 0 );
                     var classeFim = document.all.classeFinal.value.charCodeAt( 0 );
                     if( classeIni > classeFim ){
                         alert( 'Classe inicial maior que final' );
                         document.all.classeInicial.select();
                         return false;
                     }
                     Disabled( "blocoConsolidacao", false );
                     return true;
                  }
